     2. Parsear los datos que llegan desde el frontend
    const data = JSON.parse(e.postData.contents);
    
     3. Validar seguridad
    if (!data.key  data.key !== SECRET_API_KEY) {
      return ContentService

        .setMimeType(ContentService.MimeType.JSON);
    }
    
    if (!sheet) {
      return ContentService

        .setMimeType(ContentService.MimeType.JSON);
    }
    
     4. Calcular el ID automático (número de fila siguiente o contador)
    const lastRow = sheet.getLastRow();

     5. Preparar los datos a insertar
    const timestamp = new Date();
    const userId = data.user_id  guest_unknown;
    const eventName = data.event_name  unknown_event;
    const errorDetail = data.error_detail  ;
    
     6. Escribir la fila en la solapa Logs
    sheet.appendRow([eventId, userId, eventName, errorDetail, timestamp]);
